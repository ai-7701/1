7 Merge Sort

Aim

To sort an array using divide and conquer by splitting it in halves, sorting each half and merging them.

Algorithm

  1. If l < r, find the middle m = (l+r)/2.
  2. Recursively sort the left half a[l..m].
  3. Recursively sort the right half a[m+1..r].
  4. Merge the two sorted halves into a temp array using two pointers.
  5. Copy the temp array back into a[l..r].

Code

Original code (q7.c)
#include<stdio.h>

void merge(int a[],int l,int m,int r){

    int i=l;
    int j=m+1;
    int k=l;
    int temp[100];
    while(i<=m && j<=r){
        if(a[i]<=a[j]){
           temp[k]=a[i];
              i++;
              k++;
        }else{
            temp[k]=a[j];
            k++;
            j++;
        }
    }
    while(i<=m){
        temp[k]=a[i];
        i++;
        k++;
    }
    while(j<=r){
        temp[k]=a[j];
        k++;
        j++;
    }

    for(int i=l;i<=r;i++){
        a[i]=temp[i];
    }


}

void mergesort(int a[],int l, int r){
   if(l<r){    
    int m=(l+r)/2;

    mergesort(a,l,m);
    mergesort(a,m+1,r);
    merge(a,l,m,r);

   }
    
}

int main(){
    int a[]={5,3,8,2,1};
    int n=sizeof(a)/sizeof(a[0]);

    printf("mergesort: ");
    mergesort(a,0,n-1);
    for(int i=0;i<n;i++){
        printf("%d ",a[i]);
    }
}

Recurrence Relation

T(n) = 2T(n/2) + n, T(1) = 1 ⇒ T(n) = O(n log n) (Master theorem, case 2)

Generalised Code (Short Version)

Generalised short version
#include<stdio.h>
void ms(int a[],int l,int r){
    if(l>=r) return;
    int m=(l+r)/2,t[100],i=l,j=m+1,k=l;
    ms(a,l,m); ms(a,m+1,r);
    while(i<=m && j<=r) t[k++] = a[i]<=a[j] ? a[i++] : a[j++];
    while(i<=m) t[k++]=a[i++];
    while(j<=r) t[k++]=a[j++];
    for(k=l;k<=r;k++) a[k]=t[k];
}
int main(){
    int a[]={5,3,8,2,1},n=5;
    ms(a,0,n-1);
    for(int i=0;i<n;i++) printf("%d ",a[i]);
}
